cmake_minimum_required(VERSION 3.24)
project(BWPM VERSION 0.1.0 DESCRIPTION "Fast bidirectional vector wave propagation method" LANGUAGES NONE)

option(BWPM_BUILD_CORE "Build the compiled module bwpm._core" ON)
option(BWPM_ENABLE_CUDA "Build the CUDA backend (RTX 6000 Ada, sm_89), stage B4d" OFF)
option(BWPM_CUFFT_STATIC "Link cuFFT statically (larger module, no run-time library path needed)" OFF)
option(BWPM_STATIC_LIBSTDCXX "Link libstdc++ and libgcc statically into _core (GCC; for conda Pythons with an old libstdc++)" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT BWPM_BUILD_CORE)
  message(STATUS "BWPM: core disabled, pure Python reference only (no compiler needed)")
  return()
endif()
enable_language(CXX)

add_library(bwpm_core STATIC cpp/src/interface.cpp cpp/src/solver.cpp)
target_include_directories(bwpm_core PUBLIC cpp/include)
target_include_directories(bwpm_core SYSTEM PRIVATE cpp/third_party/pocketfft)
find_package(Threads REQUIRED)
target_link_libraries(bwpm_core PUBLIC Threads::Threads)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(bwpm_core PUBLIC OpenMP::OpenMP_CXX)
endif()
set_target_properties(bwpm_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(bwpm_core PRIVATE /W4)
else()
  target_compile_options(bwpm_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BWPM_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89)  # RTX 6000 Ada (decision E7, D8)
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit 12.6 REQUIRED)
  add_library(bwpm_cuda STATIC cpp/src/solver_cuda.cu)
  target_include_directories(bwpm_cuda PUBLIC cpp/include)
  # static CUDA runtime; cuFFT either static or found at run time through the RPATH of _core
  if(BWPM_CUFFT_STATIC)
    target_link_libraries(bwpm_cuda PUBLIC CUDA::cudart_static CUDA::cufft_static)
  else()
    target_link_libraries(bwpm_cuda PUBLIC CUDA::cudart_static CUDA::cufft)
    # RPATH: directory of the cuFFT library that was linked (symlink and resolved real path)
    get_target_property(_bwpm_cufft CUDA::cufft IMPORTED_LOCATION)
    if(NOT _bwpm_cufft)
      get_target_property(_bwpm_cufft CUDA::cufft IMPORTED_LOCATION_RELEASE)
    endif()
    set(BWPM_CUDA_RPATH "${CUDAToolkit_LIBRARY_DIR}")
    if(_bwpm_cufft)
      get_filename_component(_bwpm_dir "${_bwpm_cufft}" DIRECTORY)
      file(REAL_PATH "${_bwpm_cufft}" _bwpm_real)
      get_filename_component(_bwpm_real_dir "${_bwpm_real}" DIRECTORY)
      list(APPEND BWPM_CUDA_RPATH "${_bwpm_dir}" "${_bwpm_real_dir}")
      list(REMOVE_DUPLICATES BWPM_CUDA_RPATH)
    endif()
    message(STATUS "BWPM CUDA: cuFFT ${_bwpm_cufft}; RPATH ${BWPM_CUDA_RPATH}")
  endif()
  set_target_properties(bwpm_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()

if(SKBUILD)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/python/bindings.cpp)
  target_link_libraries(_core PRIVATE bwpm_core)
  if(BWPM_ENABLE_CUDA)
    target_link_libraries(_core PRIVATE bwpm_cuda)
    target_compile_definitions(_core PRIVATE BWPM_HAVE_CUDA)
    if(NOT BWPM_CUFFT_STATIC)
      set_target_properties(_core PROPERTIES BUILD_RPATH "${BWPM_CUDA_RPATH}"
                                             INSTALL_RPATH "${BWPM_CUDA_RPATH}"
                                             INSTALL_RPATH_USE_LINK_PATH ON)
    endif()
  endif()
  if(BWPM_STATIC_LIBSTDCXX)
    target_link_options(_core PRIVATE -static-libstdc++ -static-libgcc)
  endif()
  install(TARGETS _core LIBRARY DESTINATION bwpm)
endif()

message(STATUS "BWPM ${PROJECT_VERSION}: core=${BWPM_BUILD_CORE} CUDA=${BWPM_ENABLE_CUDA}")
